$f="<file>"; $t=[System.IO.File]::ReadAllText($f)
$s=$t.IndexOf("متن تألیفی"); $e=$t.IndexOf("تصویرگری")
(($t.Substring($s,$e-$s) -split '\s+') | Where-Object {$_ -ne ''} | Measure-Object).Count
